🧮 Lógica de Cálculo (calculations.js) — 2026 Rates

js
export const calculatePayroll = (salaryBrut) => {
  const rates = {
    maladieSoins: 0.028,  // 2.8%
    maladieEspeces: 0.0025, // 0.25%
    pension: 0.08,  // 8.0%
    dependance: 0.014, // 1.4%
    impots: 0.0842, // média 2026, ajustável via tabela
  };

  const cotisations = salaryBrut * (rates.maladieSoins + rates.maladieEspeces + rates.pension);
  const dependance = salaryBrut * rates.dependance;
  const totalSocial = cotisations + dependance;

  const totalImposable = salaryBrut - totalSocial;
  const impots = totalImposable * rates.impots;
  const credit = 58; // exoneração fixa exemplo

  const net = totalImposable - impots + credit;
  return { cotisations, dependance, totalSocial, totalImposable, impots, credit, net };
};

⚙️ PARTE 3 — LÓGICA DE CÁLCULO (2026)
Cria frontend/src/utils/calculations.js:

js
export const calculateLuxSalary = (salaryBrut) => {
  const rates = {
    maladieSoins: 0.028,        // 2.80%
    maladieEspeces: 0.0025,     // 0.25%
    pension: 0.08,              // 8.00%
    dependance: 0.014,          // 1.40%
  };

  const cotisations = salaryBrut * (rates.maladieSoins + rates.maladieEspeces + rates.pension);
  const dependance = salaryBrut * rates.dependance;
  const totalSocial = cotisations + dependance;
  const totalImposable = salaryBrut - totalSocial;

  // Placeholder médio de imposto (ajustável por escalão)
  const impotsRate = 0.0842; 
  const impots = totalImposable * impotsRate;
  const credit = 58; // Correção fiscal média

  const net = totalImposable - impots + credit;

  return {
    cotisations: cotisations.toFixed(2),
    dependance: dependance.toFixed(2),
    totalSocial: totalSocial.toFixed(2),
    totalImposable: totalImposable.toFixed(2),
    impots: impots.toFixed(2),
    credit,
    net: net.toFixed(2)
  };
};

📤 PARTE 6 — PDF E EXPORTAÇÕES
Cria frontend/src/utils/generatePDF.js com jsPDF:

js
import { jsPDF } from "jspdf";

export const generatePayslipPDF = (employee, results) => {
  const doc = new jsPDF();
  doc.setFontSize(12);
  doc.text("LUXPAYROLL 2026 - Fiche de Salaire", 20, 20);
  doc.text(`Employé: ${employee.name || "N/A"}`, 20, 40);
  doc.text(`Salaire brut: €${results.totalImposable}`, 20, 60);
  doc.text(`Cotisations sociales: €${results.totalSocial}`, 20, 80);
  doc.text(`Impôts: €${results.impots}`, 20, 100);
  doc.text(`Net à payer: €${results.net}`, 20, 120);
  doc.save(`Fiche_Salaire_${employee.name || "Employe"}.pdf`);
};
🧾 PARTE 7 — BACKEND (API REST)
Dentro do diretório backend/:

Cria server.js:

js
import express from "express";
import cors from "cors";
import bodyParser from "body-parser";

const app = express();
app.use(cors());
app.use(bodyParser.json());

// Endpoint cálculo
app.post("/api/calculate", (req, res) => {
  const { salaryBrut } = req.body;
  const rates = {
    maladieSoins: 0.028,
    maladieEspeces: 0.0025,
    pension: 0.08,
    dependance: 0.014,
    impots: 0.0842,
  };
  const cotisations = salaryBrut * (rates.maladieSoins + rates.maladieEspeces + rates.pension);
  const dependance = salaryBrut * rates.dependance;
  const totalSocial = cotisations + dependance;
  const totalImposable = salaryBrut - totalSocial;
  const impots = totalImposable * rates.impots;
  const credit = 58;
  const net = totalImposable - impots + credit;

  res.json({ cotisations, dependance, totalSocial, totalImposable, impots, credit, net });
});

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => console.log(`✅ Backend running on ${PORT}`));
